cmake_minimum_required(VERSION 3.20)
project(4agent VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "4agent targets Linux. Configure and test inside WSL on Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(agent_core src/core.cpp)
target_include_directories(agent_core PUBLIC src)
target_compile_options(agent_core PRIVATE -Wall -Wextra -Wpedantic)
set(CURSES_NEED_WIDE TRUE)
find_package(Curses REQUIRED)
add_executable(4agent src/main.cpp src/admin.cpp src/tui.cpp)
target_include_directories(4agent PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(4agent PRIVATE agent_core ${CURSES_LIBRARIES})
target_compile_definitions(4agent PRIVATE AGENT_INSTALL_PATH="${CMAKE_INSTALL_PREFIX}/sbin/4agent")
target_compile_options(4agent PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS 4agent RUNTIME DESTINATION sbin)
configure_file(systemd/4agent-expire.service.in 4agent-expire.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/4agent-expire.service systemd/4agent-expire.timer DESTINATION lib/systemd/system)
install(FILES README.md DESTINATION share/doc/4agent)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE agent_core)
add_test(NAME core COMMAND core_tests)

option(FOURAGENT_ENABLE_PACKAGING "Enable Debian package generation (requires /usr install prefix)" OFF)
if(FOURAGENT_ENABLE_PACKAGING)
  if(NOT CMAKE_INSTALL_PREFIX STREQUAL "/usr")
    message(FATAL_ERROR "Debian packages require -DCMAKE_INSTALL_PREFIX=/usr")
  endif()
  set(CPACK_GENERATOR DEB)
  set(CPACK_PACKAGE_NAME "4agent")
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(FOURAGENT_DEB_VERSION "${PROJECT_VERSION}-1" CACHE STRING "Debian package version")
  set(CPACK_DEBIAN_PACKAGE_VERSION "${FOURAGENT_DEB_VERSION}")
  set(CPACK_PACKAGE_CONTACT "hdmain <166563896+hdmain@users.noreply.github.com>")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Terminal UI for temporary AI agent accounts")
  set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/hdmain/4agent")
  set(CPACK_DEBIAN_PACKAGE_SECTION "admin")
  set(CPACK_DEBIAN_PACKAGE_PRIORITY "optional")
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "openssh-server, openssh-client, sudo, curl, procps, passwd, systemd, init-system-helpers")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
  set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA
    "${CMAKE_CURRENT_SOURCE_DIR}/packaging/debian/postinst"
    "${CMAKE_CURRENT_SOURCE_DIR}/packaging/debian/prerm"
    "${CMAKE_CURRENT_SOURCE_DIR}/packaging/debian/postrm")
  include(CPack)
endif()
